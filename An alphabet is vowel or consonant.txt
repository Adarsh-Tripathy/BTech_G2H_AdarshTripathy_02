#include <stdio.h>
int main(){
    char a, b[]="aeiouAEIOU";
    int c=0;
    printf("Enter a alphabet to check it is vowel or constant:");
    scanf("%c",&a);
    for(int i=0;i<10;i++){
        if(a==b[i]){
            printf(" Alphabet is vowel");
            break;
        }
        else c=c+1;
    }
    if(c==10){
        printf("Alphabet is consonant");
    }
   }
