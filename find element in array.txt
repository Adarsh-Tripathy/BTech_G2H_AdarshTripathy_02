#include<stdio.h>
int main(){
    int n,a,c=0;
    printf("enter the number of element in array:");
    scanf("%d",&n);
    int arr[n];
    for(int i=0;i<n;i++){
        scanf("%d",&arr[i]);
    }
    printf("enter the number to search:");
    scanf("%d",&a);
    for(int i=0;i<n;i++){
        if(arr[i]==a){
            printf("found at index %d",i);
            break;
        }
        else c+=1;
    }
    if(c==n) {
        printf("not found");
    }
}
