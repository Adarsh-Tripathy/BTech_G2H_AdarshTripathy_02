#include<stdio.h>
int main(){
    int a,b,c;
    printf("Enter two numbers:");
    scanf("%d%d",&a,&b);
    printf("enter 1,2,3,4 for *,/,+,-\n");
    scanf("%d",&c);
    switch(c){
        case 1:
            printf("%d",a*b);
            break;
        case 2:
            printf("%d",a/b);
            break;
        case 3:
            printf("%d",a+b);
            break;
        case 4:
            printf("%d",a-b);
            break;
        default:
            printf("Invalid");
    }
    return 0;
}
